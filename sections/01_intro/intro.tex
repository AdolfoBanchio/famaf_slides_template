\section{Introducción}

\begin{frame}{Introducción}
  \begin{block}{Objetivo}
    Presentar una idea clara, estructurada y visualmente ordenada.
  \end{block}

  \vspace{0.3cm}
  \begin{itemize}
    
    \item Mantener una narrativa simple.
    \item Usar pocas palabras por diapositiva.
    \item Priorizar el contenido visual y el contraste.
  \end{itemize}
\end{frame}

\begin{frame}{Estructura recomendada}
  \begin{itemize}
    \item Título claro
    \item Una idea central por slide
    \item Listas cortas y consistentes
    \item Un bloque o gráfico cuando haga falta
  \end{itemize}
\end{frame}
